GIRVAN-NEWMAN COMMUNITY DETECTION IN R

# Install package if required:
# install.packages("igraph")

library(igraph)

# Create a sample social network
edges <- c(
  "A","B",
  "A","C",
  "B","C",
  "B","D",
  "C","D",
  "E","F",
  "E","G",
  "F","G",
  "D","E"
)

g <- graph(edges = edges, directed = FALSE)

# Display graph
plot(
  g,
  vertex.color = "lightblue",
  vertex.size = 30,
  vertex.label.color = "black",
  main = "Girvan-Newman Community Detection"
)

# Girvan-Newman algorithm
communities <- cluster_edge_betweenness(g)

# Display communities
print(communities)

# Membership of each vertex
print(membership(communities))

# Number of communities
print(length(communities))

# Modularity
print(modularity(communities))

# Plot graph with detected communities
plot(
  communities,
  g,
  main = "Communities Detected using Girvan-Newman"
)
