FIALET-MARTIN ALGORITHM (DISTINCT ELEMENT ESTIMATION)

# Estimates the number of distinct elements in a data stream
# using hashing and the position of the least significant 1-bit.

def fialet_martin(stream):
    max_zero_position = 0

    for item in stream:
        value = hash(item)
        value = abs(value)

        if value == 0:
            continue

        # Position of the rightmost set bit (1), starting from 0
        position = 0
        while (value & 1) == 0:
            position += 1
            value >>= 1

        max_zero_position = max(max_zero_position, position)

    # Estimated number of distinct elements
    return 2 ** max_zero_position


if __name__ == "__main__":
    data = [
        "apple", "banana", "apple", "orange",
        "banana", "mango", "grapes", "orange"
    ]

    actual = len(set(data))
    estimate = fialet_martin(data)

    print("Data:", data)
    print("Actual distinct elements:", actual)
    print("Fialet-Martin estimated distinct elements:", estimate)
