HDFS BASICS AND FILE OPERATIONS

# Start Hadoop services (Linux)
start-dfs.sh
start-yarn.sh

# Check HDFS
hdfs dfs -ls /

# Create a directory in HDFS
hdfs dfs -mkdir -p /user/student/input

# 1. Copy a local file to HDFS
hdfs dfs -put sample.txt /user/student/input/

# Alternative
hdfs dfs -copyFromLocal sample.txt /user/student/input/

# 2. Copy a file from HDFS to local file system
hdfs dfs -get /user/student/input/sample.txt ./sample_copy.txt

# Alternative
hdfs dfs -copyToLocal /user/student/input/sample.txt ./sample_copy.txt

# 3. Delete a file from HDFS
hdfs dfs -rm /user/student/input/sample.txt

# Delete a directory recursively
hdfs dfs -rm -r /user/student/input

# 4. Move a file in HDFS
hdfs dfs -mkdir -p /user/student/output
hdfs dfs -put sample.txt /user/student/input/
hdfs dfs -mv /user/student/input/sample.txt /user/student/output/

# Display/list files
hdfs dfs -ls /user/student/output

# Display file contents
hdfs dfs -cat /user/student/output/sample.txt

# Stop Hadoop services
stop-yarn.sh
stop-dfs.sh
